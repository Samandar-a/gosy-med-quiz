rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /quizDeviceLocks/{appId} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['deviceHash', 'createdAt'])
                    && request.resource.data.deviceHash is string;
      allow update: if false;
      allow delete: if false;
    }
  }
}
